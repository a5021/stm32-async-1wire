88// Main macro: expands to a mask of all requested bitfields for a given peripheral register
89#define BITS(PREFIX_, REG, ...) \
90( \
91 (void)sizeof(struct { \
92 _Static_assert(sizeof(#__VA_ARGS__) > 1, \
93 "BITS(): at least one bitfield must be provided"); \
94 int _dummy; \
95 }), \
96 _BITS_CHOOSER(_BITS_NARG(__VA_ARGS__))(PREFIX_, REG, __VA_ARGS__) \
97)
