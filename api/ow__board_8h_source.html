34/* ------------------------------------------------------------------
35 * Family selection: a single OW_FAMILY_* token resolved from
36 * the CMSIS device macro of the target (e.g. -DSTM32F030x6) or, for
37 * host-test mocks, PlatformIO and STM32CubeMX projects, from a family
38 * macro (STM32F0, STM32F1, STM32F3, STM32F4, STM32G0, STM32G4, STM32H5,
39 * STM32H7). Concrete device spellings are accepted for the parts this project
40 * supports (STM32F030x6 for F0; STM32F103xB for F1; STM32F303xC for F3;
41 * STM32F407xx / STM32F401xC / STM32F401xE / STM32F446xx for F4;
42 * STM32G031xx for G0; STM32G474xx for G4; STM32H503xx for H5;
43 * STM32H743xx for H7).
44 * ------------------------------------------------------------------ */
45#if defined(STM32F030x6) || defined(STM32F0)
46#define OW_FAMILY_F0
47#elif defined(STM32F103xB) || defined(STM32F1)
48#define OW_FAMILY_F1
49#elif defined(STM32F303xC) || defined(STM32F3)
50#define OW_FAMILY_F3
51#elif defined(STM32F407xx) || defined(STM32F401xC) || defined(STM32F401xE) || \
52 defined(STM32F446xx) || defined(STM32F4)
53#define OW_FAMILY_F4
54#elif defined(STM32G031xx) || defined(STM32G0)
55#define OW_FAMILY_G0
56#elif defined(STM32G474xx) || defined(STM32G4)
57#define OW_FAMILY_G4
58#elif defined(STM32H503xx) || defined(STM32H5)
59#define OW_FAMILY_H5
60#elif defined(STM32H743xx) || defined(STM32H7)
61#define OW_FAMILY_H7

The Make/CMake builds supply it from OW_MCU/chips/<part>.mk."

82/* ------------------------------------------------------------------
83 * Board clock defaults. The external crystal belongs to the board and
84 * the system clock to the application, so they are separate knobs; the
85 * build passes OW_SYSCLK_MHZ for the part, and app.c reads these.
86 * A wrong OW_HSE_MHZ is not a compile error - the PLL simply never
87 * locks. The *_MAX_SYSCLK_MHZ ceilings let app.c reject a request
88 * above what the concrete part can take.
89 * ------------------------------------------------------------------ */
90#if defined(OW_FAMILY_F3)
91/* No HSI16 on this family: the 72MHz PLL multiplies the crystal by 9. */
92#if !defined(OW_HSE_MHZ)
93#define OW_HSE_MHZ 8
94#endif
95#elif defined(OW_FAMILY_F4)
96#if !defined(OW_F4_MAX_SYSCLK_MHZ)
97#if defined(STM32F446xx)
98#define OW_F4_MAX_SYSCLK_MHZ 180 /* F446: the only part here that needs the over-drive sequence */
99#elif defined(STM32F401xC) || defined(STM32F401xE)
100#define OW_F4_MAX_SYSCLK_MHZ 84 /* F401 parts top out at 84MHz */
101#else
102#define OW_F4_MAX_SYSCLK_MHZ 168 /* F405/F407 cap */
103#endif
104#endif
105#if !defined(OW_HSE_MHZ)
106#define OW_HSE_MHZ 8
107#endif
108#elif defined(OW_FAMILY_G4)
109#if !defined(OW_G4_MAX_SYSCLK_MHZ)
110#define OW_G4_MAX_SYSCLK_MHZ 170 /* G474 ceiling, voltage Range 1 Boost */
111#endif
112#if !defined(OW_HSE_MHZ)
113#define OW_HSE_MHZ 8 /* WeAct STM32G474CBT6 Long: 8MHz crystal */
114#endif
115#elif defined(OW_FAMILY_H5)
116#if !defined(OW_H5_MAX_SYSCLK_MHZ)
117#define OW_H5_MAX_SYSCLK_MHZ 250 /* H503 ceiling */
118#endif
119#if !defined(OW_HSE_MHZ)
120#define OW_HSE_MHZ 8 /* WeAct STM32H503Cx Core Board: 8MHz crystal (Y2) */
121#endif
122#elif defined(OW_FAMILY_H7)
123#if !defined(OW_H7_MAX_SYSCLK_MHZ)
124#define OW_H7_MAX_SYSCLK_MHZ 480 /* Rev V H743 core ceiling; 480MHz requires VOS0 */
125#endif
126#if !defined(OW_H7_MAX_TIM1_MHZ)
127#define OW_H7_MAX_TIM1_MHZ 240 /* APB2 <=120MHz, timer x2; this is the 1-Wire timing ceiling */
128#endif
129#if !defined(OW_HSE_MHZ)
130#define OW_HSE_MHZ 25 /* WeAct Studio STM32H7XX board: 25MHz crystal */
131#endif
132#endif
