84/* ------------------------------------------------------------------
85 * Family selection: a single OW_FAMILY_* token resolved from
86 * the CMSIS device macro of the target (e.g. -DSTM32F030x6) or, for
87 * host-test mocks, PlatformIO and STM32CubeMX projects, from a family
88 * macro (STM32F0, STM32F1, STM32F3, STM32F4, STM32G0, STM32G4, STM32L4,
89 * STM32H5, STM32H7). Concrete device spellings are accepted for the parts this project
90 * supports (STM32F030x6 for F0; STM32F103xB for F1; STM32F303xC for F3;
91 * STM32F407xx / STM32F401xC / STM32F446xx for F4;
92 * STM32G031xx for G0; STM32G474xx for G4; STM32L476xx for L4;
93 * STM32H503xx for H5; STM32H743xx for H7).
94 * ------------------------------------------------------------------ */

96#define OW_FAMILY_F0
97#elif defined(STM32F103xB) || defined(STM32F1)
98#define OW_FAMILY_F1
99#elif defined(STM32F303xC) || defined(STM32F3)
100#define OW_FAMILY_F3
101#elif defined(STM32F407xx) || defined(STM32F401xC) || \
102 defined(STM32F446xx) || defined(STM32F4)
103#define OW_FAMILY_F4
104#elif defined(STM32G031xx) || defined(STM32G0)
105#define OW_FAMILY_G0
106#elif defined(STM32G474xx) || defined(STM32G4)
107#define OW_FAMILY_G4
108#elif defined(STM32L476xx) || defined(STM32L4)

112#elif defined(STM32H503xx) || defined(STM32H5)

135/* ------------------------------------------------------------------
136 * Board clock defaults. The external crystal belongs to the board and
137 * the system clock to the application, so they are separate knobs; the
138 * build passes OW_SYSCLK_MHZ for the part, and app.c reads these.
139 * A wrong OW_HSE_MHZ is not a compile error - the PLL simply never
140 * locks. The *_MAX_SYSCLK_MHZ ceilings let app.c reject a request
141 * above what the concrete part can take.
142 * ------------------------------------------------------------------ */
143#if defined(OW_FAMILY_F3)
144/* No HSI16 on this family: the 72MHz PLL multiplies the crystal by 9. */
145#if !defined(OW_HSE_MHZ)
146#define OW_HSE_MHZ 8
147#endif
148#elif defined(OW_FAMILY_F4)
149#if !defined(OW_F4_MAX_SYSCLK_MHZ)
150#if defined(STM32F446xx)
151#define OW_F4_MAX_SYSCLK_MHZ 180 /* F446: the only part here that needs the over-drive sequence */
152#elif defined(STM32F401xC)
153#define OW_F4_MAX_SYSCLK_MHZ 84 /* F401 parts top out at 84MHz */
154#else
155#define OW_F4_MAX_SYSCLK_MHZ 168 /* F405/F407 cap */
156#endif
157#endif
158#if !defined(OW_HSE_MHZ)
159#define OW_HSE_MHZ 8
160#endif
161#elif defined(OW_FAMILY_G4)
162#if !defined(OW_G4_MAX_SYSCLK_MHZ)
163#define OW_G4_MAX_SYSCLK_MHZ 170 /* G474 ceiling, voltage Range 1 Boost */
164#endif
165#if !defined(OW_HSE_MHZ)
166#define OW_HSE_MHZ 8 /* WeAct STM32G474CBT6 Long: 8MHz crystal */
167#endif
168#elif defined(OW_FAMILY_L4)
169#if !defined(OW_L4_MAX_SYSCLK_MHZ)
170#define OW_L4_MAX_SYSCLK_MHZ 80 /* L476 ceiling, voltage Range 1 */
171#endif
172/* No OW_HSE_MHZ default: the NUCLEO-L476RG has no crystal populated and this
173 * backend is HSI16-only by design, so there is no external-oscillator knob to
174 * get wrong. */
175#elif defined(OW_FAMILY_H5)
176#if !defined(OW_H5_MAX_SYSCLK_MHZ)
177#define OW_H5_MAX_SYSCLK_MHZ 250 /* H503 ceiling */
178#endif
179#if !defined(OW_HSE_MHZ)
180#define OW_HSE_MHZ 8 /* WeAct STM32H503Cx Core Board: 8MHz crystal (Y2) */
181#endif
182#elif defined(OW_FAMILY_C5)
183#if !defined(OW_C5_MAX_SYSCLK_MHZ)
184#define OW_C5_MAX_SYSCLK_MHZ 144 /* C552 ceiling */
185#endif
186#if !defined(OW_HSE_MHZ)
187#define OW_HSE_MHZ 8 /* WeAct STM32C5 48-pin Core Board: 8MHz crystal (Y2) */
188#endif
189#elif defined(OW_FAMILY_H7)
190#if !defined(OW_H7_MAX_SYSCLK_MHZ)
191#define OW_H7_MAX_SYSCLK_MHZ 480 /* Rev V H743 core ceiling; 480MHz requires VOS0 */
192#endif
193#if !defined(OW_H7_MAX_TIM1_MHZ)
194#define OW_H7_MAX_TIM1_MHZ 240 /* APB2 <=120MHz, timer x2; this is the 1-Wire timing ceiling */
195#endif
196#if !defined(OW_HSE_MHZ)
197#define OW_HSE_MHZ 25 /* WeAct Studio STM32H7XX board: 25MHz crystal */
198#endif
199#endif
