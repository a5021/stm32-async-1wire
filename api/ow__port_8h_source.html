38/* --- CH4 input-capture digital filter (IC4F), one standard for every clock.
39 * Keep the filter time T_f = N × T_sample as close to ~500ns as the
40 * discrete IC4F table allows for the configured clock: that rejects
41 * sub-µs bus glitches while adding well under 1µs to read-slot captures
42 * — negligible against the ONEWIRE_SHORT_PULSE_MAX decode window (an
43 * IC4F sweep on STM32F030@8MHz decoded cleanly from fCK_INT N=2 all the
44 * way to fDTS/4 N=8):
45 * ≤ 8MHz fCK_INT, N=4 T_f ≈ 500ns @ 8MHz
46 * ≤16MHz fCK_INT, N=8 T_f ≈ 500ns @ 16MHz
47 * >16MHz fDTS/4, N=8 T_f ≈ 444..667ns @ 48..72MHz
48 * Backends feed the macro into TIM_CCMR2(...) unchanged. --- */
49#if (OW_PORT_SYSCLK_MHZ) <= 8
50#define OW_PORT_IC4F_ARGS IC4F_1 /* fCK_INT, N=4 */
