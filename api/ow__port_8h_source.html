39/* The byte-read capture path (ow_port_read_data, width==8) stores CCR4's
40 * least-significant byte via MSIZE=8. This is lossless only while every

47/* --- CH4 input-capture digital filter (IC4F), one standard for every clock.
48 * Keep the filter time T_f = N × T_sample as close to ~500ns as the
49 * discrete IC4F table allows for the configured clock: that rejects
50 * sub-µs bus glitches while adding well under 1µs to read-slot captures
51 * — negligible against the ONEWIRE_SHORT_PULSE_MAX decode window (an
52 * IC4F sweep on STM32F030@8MHz decoded cleanly from fCK_INT N=2 all the
53 * way to fDTS/4 N=8):
54 * ≤ 8MHz fCK_INT, N=4 T_f ≈ 500ns @ 8MHz
55 * ≤16MHz fCK_INT, N=8 T_f ≈ 500ns @ 16MHz
56 * ≤72MHz fDTS/4, N=8 T_f ≈ 444..667ns @ 48..72MHz
57 * >72MHz fDTS/8, N=6 T_f ≈ 571ns @ 84MHz
58 * (fDTS/4,N=8 on 84MHz would give 381ns — below the 444..667ns T_f range
59 * shared by every other port; fDTS/8,N=6 restores it.)
60 * Backends feed the macro into TIM_CCMR2(...) unchanged.
61 * The default can be overridden from the build (-DOW_PORT_IC4F_ARGS=...)
62 * to sweep the filter on a bench; the exact ICxF encoding is family
63 * specific (F4/TIM1 fDTS table from STM32Cube LL, e.g. fDTS/4,N=8 =
64 * IC4F_0,IC4F_1,IC4F_2 vs. fDTS/8,N=6 = IC4F_3). --- */
65#ifndef OW_PORT_IC4F_ARGS
66#if (OW_PORT_SYSCLK_MHZ) <= 8
67#define OW_PORT_IC4F_ARGS IC4F_1 /* fCK_INT, N=4 */
68#elif (OW_PORT_SYSCLK_MHZ) <= 16
69#define OW_PORT_IC4F_ARGS IC4F_0, IC4F_1 /* fCK_INT, N=8 */
70#elif (OW_PORT_SYSCLK_MHZ) <= 72
71#define OW_PORT_IC4F_ARGS IC4F_0, IC4F_1, IC4F_2 /* fDTS/4, N=8 */
72#else
73#define OW_PORT_IC4F_ARGS IC4F_3 /* fDTS/8, N=6 */
74#endif
75#endif
