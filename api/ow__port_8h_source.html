65/* The byte-read capture path (ow_port_read_data, width==8) stores CCR4's
66 * least-significant byte via the DMA memory-store width. How the offsets are
67 * expressed differs per backend: the basic-DMA backends keep PSIZE=16 (CCR4 is
68 * a 16-bit halfword) and let MSIZE=8 store only the low byte, while the F4
69 * backend runs direct mode with PSIZE=MSIZE=8, so the DMA fetches just the low
70 * byte of CCR4. Both are lossless only while every slot capture stays below
71 * 256 microseconds: the counter runs 0..ARR, so the maximum capture value is
72 * ONEWIRE_ONE_PULSE + ONEWIRE_ZERO_PULSE + ONEWIRE_GUARD_BAND. Enforce this
73 * globally across all backends. */

77/* --- OC3 output-compare preload (OC3PE) knob ---
78 * The preload bit in TIM_CCMR2 keeps the active CCR3 stable through a slot:
79 * with OC3PE the value written before CEN only enters the output at the
80 * terminal update event, which is the "line released to idle HIGH in
81 * hardware" guarantee (see README, "Bus Idle Behaviour"). The write/capture
82 * paths that rely on it (capture, read_pair, single-slot write) gate the bit
83 * behind this flag so a bench can sweep OC3PE on/off with
84 * -DOW_PORT_OC3PE=0/-DOW_PORT_OC3PE=1 without editing code. The DMA-fed paths
85 * (feed, write_then_read) leave it off unconditionally: their reload must act
86 * immediately, which preload would break.
87 *
88 * Default 1 (preload on) identical to the historical hard-coded behaviour.
89 *
90 * OW_PORT_OC3PE_ARGS (the CCMR2 bit value OR-ed into the mask at the affected
91 * call sites) is defined with the engine below, by the same #ifndef pattern
92 * as OW_PORT_IC4F_ARGS. */

108/* ============================================================
109 * The TIM1 + DMA core shared by all seven families
110 *
111 * The 1-Wire bus on all seven of these families is the same machine: TIM1 in
112 * one-pulse mode drives the slot pulse on CH3, CH4 captures the bus in
113 * indirect mode, and a plain CH2 compare at ONE+ZERO us is the end-of-slot
114 * marker that triggers the feed DMA into CCR3. Those 17 functions are one
115 * piece of code, not four that happen to look alike - they were four, because
116 * each backend was written against its own reference manual and then never
117 * merged.
118 *
119 * What actually differs between families is small enough to name: the clock
120 * gates in ow_port_init(), the bus-pin mux, and the DMA request routing.
121 * Clocks stay per-family (OW_PORT_ENABLE_BUS_CLOCKS), and so does the pin
122 * mux: OW_PORT_CONFIG_BUS_PIN() and OW_PORT_SET_PIN_MODE() are two short
123 * statement macros per backend (F1's legacy CRH field proves the shapes are
124 * family-owned, not shared). DMA request routing defaults to a no-op (fixed-map families) and G0/G4
125 * program their DMAMUX through the same two macros.
126 *
127 * The family facts above (ow_mcu.h: ow_pulse_t, OW_PORT_SYSCLK_MHZ /
128 * SystemCoreClock, OW_PORT_TIM1_UPD_IRQn, DMA spelling) must precede this
129 * core: ow_bits.h supplies the T1/PA/RC/D1x aliases used below (including
130 * the D13/D14 channel defaults), and the family section supplies
131 * ow_pulse_t and the init hooks. Concretely the core requires:
132 *
133 * - define OW_PORT_ENABLE_BUS_CLOCKS(), which gates this family's clocks;
134 * - define OW_PORT_CONFIG_BUS_PIN() and OW_PORT_SET_PIN_MODE(push_pull),
135 * which put the bus pin into alternate-function open-drain mode and
136 * toggle it to push-pull for the parasite strong pull-up;
137 * - optionally override OW_PORT_DMA_FEED / OW_PORT_DMA_CAPTURE (default
138 * D13/D14: TIM1_CC2 feed on channel 3, TIM1_CH4 capture on channel 4)
139 * and OW_PORT_ROUTE_CAPTURE() / OW_PORT_ROUTE_FEED() (default no-ops
140 * for fixed request maps; G0 programs its DMAMUX here). The channel
141 * assignment is the one thing here that is not interchangeable - read
142 * the family section's assignment against its reference manual.
143 *
144 * There is no run-time cost to the factoring: everything stays
145 * __STATIC_FORCEINLINE and every difference is resolved by the preprocessor,
146 * so a family's object code is what it was before the split. That is checked,
147 * not assumed - the .bin of every example for the families already sharing
148 * this core is byte-identical before and after the split that introduced it.
149 *
150 * F4 rides the same core through accessor macros with DMA1 defaults (see the
151 * OW_PORT_DMA_* block below) that the F4 section overrides — same bodies,
152 * family-spelled registers, with init clocks/pins/drive arriving through
153 * the OW_PORT_ENABLE_BUS_CLOCKS/OW_PORT_CONFIG_BUS_PIN hooks — and H5
154 * through GPDMA CTR1/CTR2/REQSEL vocabulary overrides the same way. Even the
155 * merged write+read pass is shared by all seven families with no
156 * per-family knobs left: two arm-order experiments on F446 (UG-vs-DMA order,
157 * early direction-pulse arm) both behave identically, so one order serves
158 * all - see the note above ow_port_write_then_read.
159 * ============================================================ */

204/* --- CH4 input-capture digital filter (IC4F): one selection for every clock,
205 * living here because the shared core is the one place all seven families
206 * are processed. The F4 section uses this same ladder through the core
207 * (it overrides only the DMA register spelling, not the timer setup), and
208 * so does H5 (GPDMA CTR1/CTR2/REQSEL overrides only).
209 *
210 * ICxF is a 4-bit ladder. The values this port actually programs, decoded
211 * from the RM0090 table (the same one ST's stm32f0xx_ll_tim.h spells out),
212 * are:
213 *
214 * ICxF_1 = 0b0001 = fCK_INT/16, N=1
215 * ICxF_0|ICxF_1 = 0b0011 = fCK_INT/4, N=1
216 * ICxF_0|ICxF_1|ICxF_2 = 0b0111 = fCK_INT/4, N=2
217 * ICxF_3 = 0b1000 = fCK_INT/32, N=3
218 * ICxF_0|ICxF_1|ICxF_3 = 0b1011 = fCK_INT/4, N=3
219 *
220 * So the >168MHz entry is fCK_INT/4 N=3, a *shorter* window than the
221 * 168MHz entry's fCK_INT/32 N=3. That is deliberate and was validated at
222 * 180MHz on the full 7-example matrix (see port/stm32f4/HARDWARE-NOTES.md).
223 * A "~500ns target" does not exist at these kernel clocks: the ladder's
224 * finest window is fCK_INT/16 (5.6ns at 180MHz), so reasoning should start
225 * from the decoded values.
226 *
227 * Backends feed the macro into TIM_CCMR2(...) unchanged. The default can
228 * be overridden from the build (-DOW_PORT_IC4F_ARGS=...) to sweep the
229 * filter on a bench.
230 * Test: tests/test/test_timing.c::test_ic4f_matches_the_documented_tier() */
231#if defined(OW_PORT_SYSCLK_MHZ)
232#ifndef OW_PORT_IC4F_ARGS
233#if (OW_PORT_SYSCLK_MHZ) <= 8
234#define OW_PORT_IC4F_ARGS IC4F_1 /* 0b0001 = fCK_INT/16, N=1 */
235#elif (OW_PORT_SYSCLK_MHZ) <= 16
236#define OW_PORT_IC4F_ARGS IC4F_0, IC4F_1 /* 0b0011 = fCK_INT/4, N=1 */
237#elif (OW_PORT_SYSCLK_MHZ) <= 72
238#define OW_PORT_IC4F_ARGS IC4F_0, IC4F_1, IC4F_2 /* 0b0111 = fCK_INT/4, N=2 */
239#elif (OW_PORT_SYSCLK_MHZ) <= 168
240#define OW_PORT_IC4F_ARGS IC4F_3 /* 0b1000 = fCK_INT/32, N=3 */
241#else
242#define OW_PORT_IC4F_ARGS IC4F_0, IC4F_1, IC4F_3 /* 0b1011 = fCK_INT/4, N=3 */
243#endif
244#endif

264/* Prescaler for 1 us resolution: PSC = SYSCLK / 1MHz - 1, from the shared
265 * OW_PORT_SYSCLK_MHZ. INVARIANT: TIM1's kernel clock must equal SYSCLK.
266 * How each family gets there is family-specific and documented in its
267 * ow_mcu.h section: F0/F1/F3/G0/G4 leave the APB prescaler feeding TIM1 at
268 * /1, so TIM1 = PCLK = SYSCLK directly; F4 programs PPRE2=/2 (PPRE1=/4 or
269 * /2) and relies on the STM32 x2 doubling, so TIM1 = 2 x PCLK2 = SYSCLK
270 * either way. Which APB the timer sits on, and why that family's prescalers
271 * are what they are, is documented in each section; it is checked by
272 * tests/test/test_timing.c::test_apb_prescaler_div1_for_tim1(). */
273#if defined(OW_PORT_SYSCLK_MHZ)
274#define OW_PORT_TIM_PRESCALER ((OW_PORT_SYSCLK_MHZ) - 1u)
275_Static_assert(OW_PORT_TIM_PRESCALER <= 0xFFFFu,
276 "TIM prescaler exceeds 16-bit PSC register width");
277/* CCMR2 capture-mode word shared by the three capture arms below
278 * (ow_port_capture, ow_port_read_pair, ow_port_write_then_read): OC3 in PWM
279 * mode 2 on the bus pin, CC4 capturing the same pin through TI3, plus the
280 * IC4F tier above and the OC3PE knob. One macro so the three arms cannot
281 * drift; the DMA-fed merged arm drops the preload bit (see below). */
282#define OW_PORT_CCMR2_CAP TIM_CCMR2(OC3M_0, OC3M_1, OC3M_2, CC4S_1, OW_PORT_IC4F_ARGS) | OW_PORT_OC3PE_ARGS
283#define OW_PORT_CCMR2_MRG TIM_CCMR2(OC3M_0, OC3M_1, OC3M_2, CC4S_1, OW_PORT_IC4F_ARGS)
284#else /* !defined(OW_PORT_SYSCLK_MHZ): SystemCoreClock auto-path -------------
285 * For integrations that select a backend without passing a clock -D
286 * (PlatformIO/CubeMX family-macro path): the CMSIS SystemCoreClock variable
287 * supplies the frequency instead of a header fallback guess, so a CubeMX
288 * project at 64MHz can never silently inherit a 72MHz prescaler. The Make
289 * and CMake builds always pass -DOW_PORT_SYSCLK_MHZ (from chips/<part>.mk)
290 * and never reach this branch.
291 * @pre SystemCoreClock holds the real SYSCLK in Hz (SystemCoreClockUpdate()
292 * after any clock-tree change), is an integer number of MHz >= 1MHz,
293 * and TIM1's kernel clock equals SYSCLK (same invariant as above).
294 * @note The -DOW_PORT_IC4F_ARGS bench override is compile-time only and is
295 * ignored on this path; the tiers below match the ladder above.
296 * Test: tests/test/test_sysclk_auto.c */
297extern uint32_t SystemCoreClock; /* CMSIS: maintained by SystemInit()/SystemCoreClockUpdate() */
298__STATIC_FORCEINLINE uint32_t ow_port_mhz_from_hz(uint32_t hz) {
299 /* Integer-MHz truncation of the system clock. Exact for every supported
300 * clock (8/16/48/64/72/84/168/170/180/250); shifts plus one 32x32
301 * multiply, so no __aeabi_uidiv on M0/M0+. The intermediate fits 32
302 * bits (524M at 250MHz). */
303 return (uint32_t)(((hz >> 8u) * 537u) >> 21u);
304}
305__STATIC_FORCEINLINE uint16_t ow_port_psc_from_hz(uint32_t hz) {
306 const uint32_t mhz = ow_port_mhz_from_hz(hz);
307 return (uint16_t)((mhz > 0u) ? (mhz - 1u) : 0u);
308}
309__STATIC_FORCEINLINE uint32_t ow_port_ic4f_mask_from_hz(uint32_t hz) {
310 /* Same tiers as the compile-time ladder above, as a numeric CCMR2 mask. */
311 const uint32_t mhz = ow_port_mhz_from_hz(hz);
312 if (mhz <= 8u) {
313 return TIM_CCMR2_IC4F_1;
314 }
315 if (mhz <= 16u) {
316 return (uint32_t)(TIM_CCMR2_IC4F_0 | TIM_CCMR2_IC4F_1);
317 }
318 if (mhz <= 72u) {
319 return (uint32_t)(TIM_CCMR2_IC4F_0 | TIM_CCMR2_IC4F_1 | TIM_CCMR2_IC4F_2);
320 }
321 if (mhz <= 168u) {
322 return TIM_CCMR2_IC4F_3;
323 }
324 return (uint32_t)(TIM_CCMR2_IC4F_0 | TIM_CCMR2_IC4F_1 | TIM_CCMR2_IC4F_3);
325}

389/* Disable-path contract: the wait algorithm below is DMA-IP-agnostic (write

417__STATIC_FORCEINLINE void ow_port_dma_disable(volatile uint32_t* ccr) {
418 *ccr = 0; /* request disable */
419 if (*ccr & OW_PORT_DMA_EN_BIT) {
420 uint32_t guard = 1000u;
421 while ((*ccr & OW_PORT_DMA_EN_BIT) != 0u && guard-- != 0u) {
422 }
423 }

474__STATIC_FORCEINLINE uint8_t ow_port_bus_done(void) {
475 if (T1.SR & TIM_SR(UIF)) {
476 /* No software bus release needed: every operation returns the line to
477 * idle HIGH in hardware. DMA-fed writes (ow_port_feed,
478 * ow_port_write_then_read) append ONEWIRE_RELEASE_PULSE to the CCR3
479 * feed, and the direct-write/capture operations (reset, read, single
480 * slot) use an OC3PE preload of ONEWIRE_RELEASE_PULSE — both applied
481 * exactly when the one-pulse timer stops. */
482#if OW_PORT_LOW_POWER
483 /* The update event both interrupts the low-power WFE sleep and, via
484 * SEVONPEND, raises an NVIC pending bit. UIE also latches a pending
485 * bit at every ow_port_update_event() re-arm (EGR=UG). Clear the
486 * pending flag here so the next __WFE() truly sleeps; otherwise the
487 * pending bit would make __WFE() return immediately forever (silent
488 * degradation back to a busy-loop). Clearing UIF below also retires
489 * the long-stage condition read by ow_port_long_wait_pending(). */
490 NVIC_ClearPendingIRQ(OW_PORT_TIM1_UPD_IRQn);
491#endif
492 T1.SR = 0;
493 /* Retire the schedule explicitly: on hardware OPM already stopped the
494 * counter, but stating it here keeps the completion state honest
495 * without relying on that side effect (and keeps the host mock, which
496 * does not model OPM auto-stop, in agreement with hardware). */
497 T1.CR1 &= (uint32_t)~TIM_CR1(CEN);
498 return 1u;
499 }
500 return 0u;

531__STATIC_FORCEINLINE void ow_port_sleep_until_done(void) {
532 /* Prepare for sleep: clear any stale NVIC pending bit so a leftover
533 * event cannot wake the very first WFE (silent busy-loop degradation).
534 * A fresh pending bit will be latched by the timer's update when the
535 * long stage completes. */
536 NVIC_ClearPendingIRQ(OW_PORT_TIM1_UPD_IRQn);
537 /* Re-arm the event: SEV sets the event register, the first WFE returns
538 * immediately and clears it, so the second WFE in the loop truly sleeps
539 * until a new event arrives. */
540 __SEV();
541 __WFE();
542 while (!(T1.SR & TIM_SR(UIF))) {
543 __WFE(); /* sleeps; woken by the pending bit via SEVONPEND (no ISR) */
544 }
545 /* Consume the wake-up event so the next sleep starts from a clean state. */
546 NVIC_ClearPendingIRQ(OW_PORT_TIM1_UPD_IRQn);
547}
548#endif

569__STATIC_FORCEINLINE void ow_port_capture(volatile void* dst, uint16_t count, uint16_t width) {
570#if OW_DRIVE_ACTIVE
571 ow_port_set_pin_mode(0); /* read/reset phases must be open-drain (slave can pull LOW) */
572#endif
573 T1.CCMR2 = OW_PORT_CCMR2_CAP;
574 T1.CCER = TIM_CCER(CC3E, CC4E);
575#if OW_PORT_LOW_POWER
576 T1.DIER = TIM_DIER(CC4DE, UIE);
577 /* No duration decision here: ow_port_long_wait_pending() derives it from
578 * the caller's ARR/RCR, so every capture path (reset, read, long
579 * scratchpad read) shares the one OW_PORT_LONG_STAGE_US rule. */
580#else
581 T1.DIER = TIM_DIER(CC4DE);
582#endif
583 ow_port_update_event();
584 T1.CCR3 = ONEWIRE_RELEASE_PULSE;
585 OW_PORT_DMA_DISABLE_CAPTURE();
586 OW_PORT_ROUTE_CAPTURE();
587 OW_PORT_DMA_PROG_CAPTURE(dst, count, ((width == 16) ? OW_PORT_DMA_CR_RX16 : OW_PORT_DMA_CR_RX8));
588 T1.CR1 = TIM_CR1(OPM, CEN);
589}

605__STATIC_FORCEINLINE uint8_t ow_port_feed(const ow_pulse_t* cmd, uint16_t slots) {
606 if (slots == 0u || slots > ONEWIRE_MAX_SLOTS) {
607 return 0u;
608 }
609 T1.RCR = slots - 1;
610 T1.ARR = ONEWIRE_ONE_PULSE + ONEWIRE_ZERO_PULSE + ONEWIRE_GUARD_BAND;
611 T1.CCR3 = cmd[0];
612 T1.CCR2 = ONEWIRE_ONE_PULSE + ONEWIRE_ZERO_PULSE;
613 T1.CCMR2 = TIM_CCMR2(OC3M_0, OC3M_1, OC3M_2);
614 T1.CCER = TIM_CCER(CC3E);
615#if OW_PORT_LOW_POWER
616 T1.DIER = TIM_DIER(CC2DE, UIE);
617#else
618 T1.DIER = TIM_DIER(CC2DE);
619#endif
620 ow_port_update_event();
621 OW_PORT_DMA_DISABLE_FEED();
622 OW_PORT_ROUTE_FEED();
623 /* Feed slots 2..N, then the ONEWIRE_RELEASE_PULSE (bus release) */
624 OW_PORT_DMA_PROG_FEED(&cmd[1], slots, OW_PORT_DMA_CR_TX);
625 T1.CR1 = TIM_CR1(OPM, CEN);
626 return 1u;
627}

634__STATIC_FORCEINLINE void ow_port_start_timer(uint16_t arr, uint8_t rcr) {
635 T1.ARR = arr;
636 T1.RCR = rcr;
637#if OW_PORT_LOW_POWER
638 if ((uint32_t)(rcr + 1u) * (uint32_t)arr > OW_PORT_LONG_STAGE_US) {
639 /* long stage: conversion / EEPROM hold-off. Enable the update
640 * interrupt so the pending bit wakes __WFE() via SEVONPEND.
641 * ow_port_capture() already sets UIE for the capture stages, but
642 * start_timer() must too, else WFE sleeps forever. */
643 T1.DIER |= TIM_DIER(UIE);
644 }
645#endif
646 ow_port_update_event();
647 T1.CR1 = TIM_CR1(OPM, CEN);
648}

655__STATIC_FORCEINLINE void ow_port_reset(volatile uint16_t* reset_pulses) {
656 T1.RCR = 0;
657 T1.ARR = OW_PORT_RESET_TIMEOUT;
658 T1.CCR3 = OW_PORT_RESET_PULSE_DURATION;
659 /* Clear the capture buffer: only reset_pulses[0] (master release) is always
660 * written by the DMA, so a no-presence reset would otherwise leave a stale
661 * reset_pulses[1] from a previous presence reset and onewire_present() would
662 * report a false device. Zeroing makes a single-capture reset report "no
663 * device". */
664 reset_pulses[0] = 0;
665 reset_pulses[1] = 0;
666 ow_port_capture(reset_pulses, OW_PORT_CAPTURE_BUF_SIZE, 16);
667}

679__STATIC_FORCEINLINE uint8_t ow_port_write_slots(const ow_pulse_t* pulses, uint16_t slots) {
680 if (slots == 0u || slots > ONEWIRE_MAX_SLOTS) {
681 return 0u;
682 }
683#if OW_DRIVE_ACTIVE
684 ow_port_set_pin_mode(1); /* active-drive write: master drives both levels */
685#endif
686 if (slots == 1) {
687 /* Single slot: no DMA needed, avoids a zero-length DMA transaction */
688 T1.RCR = 0; /* Single slot, no repetition */
689 T1.ARR = ONEWIRE_ONE_PULSE + ONEWIRE_ZERO_PULSE + ONEWIRE_GUARD_BAND; /* Total bit slot time */
690 T1.CCR3 = pulses[0]; /* Pulse duration encodes the bit */
691 /* OC3PE plus a ONEWIRE_RELEASE_PULSE preload release the bus at the
692 * terminal update event, exactly when the one-pulse timer stops
693 * (hardware bus release). */
694 T1.CCMR2 = TIM_CCMR2(OC3M_0, OC3M_1, OC3M_2) | OW_PORT_OC3PE_ARGS;
695 T1.CCER = TIM_CCER(CC3E);
696#if OW_PORT_LOW_POWER
697 T1.DIER = TIM_DIER(UIE); /* no DMA for a single bit slot; keep UIE for WFE */
698#else
699 T1.DIER = 0; /* No DMA for a single bit slot */
700#endif
701 ow_port_update_event();
702 T1.CCR3 = ONEWIRE_RELEASE_PULSE; /* Preload release pulse -> line idles HIGH when the timer stops */
703 T1.CR1 = TIM_CR1(OPM, CEN);
704 return 1u;
705 }
706 return ow_port_feed(pulses, slots);
707}

732/* Merged-pass arm order: one order serves all families (direction pulse
733 * armed first, DMA programmed after the update event). Two per-family knobs
734 * lived here briefly and were both removed after full F446 hardware matrices
735 * proved them don't-care: DMA-after-UG behaves like DMA-before-UG, and the
736 * early direction-pulse arm behaves like arming once before CEN (14 cells
737 * each, 2026-10-06). */
745__STATIC_FORCEINLINE void ow_port_write_then_read(uint8_t bit, volatile uint16_t* pulse3,
746 const ow_pulse_t* read_pulse) {
747#if OW_DRIVE_ACTIVE
748 ow_port_set_pin_mode(0); /* merged write+read stays open-drain so the read half is safe */
749#endif
750 const ow_pulse_t write_pulse = bit ? ONEWIRE_ONE_PULSE : ONEWIRE_ZERO_PULSE;
751 OW_PORT_MARKER_TOGGLE(); /* opt-in LA hook (no-op by default): merged pass starts here */
752 T1.RCR = 2; /* Three slots, then a single update event */
753 T1.ARR = ONEWIRE_ONE_PULSE + ONEWIRE_ZERO_PULSE + ONEWIRE_GUARD_BAND; /* Total bit slot time */
754 /* Arm the direction pulse first. The bus was released idle-high by
755 * ow_port_bus_done(), so this write produces the single clean falling edge
756 * the devices re-sync their slot timer to. Holding it from the top instead
757 * of arming it right before CEN means the CC4 capture is armed while the
758 * bus is low, so the open-drain RC rise can never be mistaken for a slot
759 * edge. */
760 T1.CCR3 = write_pulse; /* Slot 1 write pulse encodes the direction bit */
761 T1.CCR2 = ONEWIRE_ONE_PULSE + ONEWIRE_ZERO_PULSE; /* End-of-slot reload trigger */
762 /* OC3 in PWM mode (no preload so the reload is immediate), CC4 capture armed */
763 T1.CCMR2 = OW_PORT_CCMR2_MRG;
764 T1.CCER = TIM_CCER(CC3E, CC4E); /* Enable both channels */
765 /* Disconnect DMA requests while re-arming the channels, then re-connect
766 * them only after the timer flags are clean and just before starting.
767 * (The end-of-slot CC2 compare event of the previous merged operation can
768 * leave a pending request that fires the reload DMA immediately on re-arm,
769 * overwriting the freshly written direction pulse in CCR3.) */
770#if OW_PORT_LOW_POWER
771 T1.DIER = TIM_DIER(UIE); /* keep UIE for WFE while the DMA requests are apart */
772#else
773 T1.DIER = 0;
774#endif
775 ow_port_update_event();
776 /* Capture DMA: write-slot capture plus the id/cmp pulse pair into the buffer.
777 * Spelled through the PROG vocabulary (DMA1 defaults above write the same
778 * CPAR/CMAR/CNDTR/CCR sequence), so a backend on different DMA IP only
779 * overrides the macros, not this body. */
780 OW_PORT_DMA_DISABLE_CAPTURE();
781 OW_PORT_ROUTE_CAPTURE();
782 OW_PORT_DMA_PROG_CAPTURE(pulse3, 3, OW_PORT_DMA_CR_RX16);
783 /* Feed DMA: reload CCR3 with the read pulse for slots 2-3, then write
784 * ONEWIRE_RELEASE_PULSE during slot 3 so the one-pulse timer stops with
785 * the line released to idle HIGH (hardware bus release). */
786 OW_PORT_DMA_DISABLE_FEED();
787 OW_PORT_ROUTE_FEED();
788 OW_PORT_DMA_PROG_FEED(read_pulse, 3, OW_PORT_DMA_CR_TX);
789#if OW_PORT_LOW_POWER
790 T1.DIER = TIM_DIER(CC4DE, CC2DE, UIE); /* Capture + CCR3 reload via DMA (UIE for WFE) */
791#else
792 T1.DIER = TIM_DIER(CC4DE, CC2DE); /* Capture + CCR3 reload via DMA */
793#endif
794 T1.CCR3 = write_pulse; /* Re-arm the direction pulse (safe against a stale CC2 DMA reload) */
795 T1.CR1 = TIM_CR1(OPM, CEN);
796}
