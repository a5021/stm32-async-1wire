131/* The byte-read capture path (ow_port_read_data, width==8) stores CCR4's
132 * least-significant byte via the DMA memory-store width. How the offsets are
133 * expressed differs per backend: the basic-DMA backends keep PSIZE=16 (CCR4 is
134 * a 16-bit halfword) and let MSIZE=8 store only the low byte, while the F4
135 * backend runs direct mode with PSIZE=MSIZE=8, so the DMA fetches just the low
136 * byte of CCR4. Both are lossless only while every slot capture stays below
137 * 256 microseconds: the counter runs 0..ARR, so the maximum capture value is
138 * ONEWIRE_ONE_PULSE + ONEWIRE_ZERO_PULSE + ONEWIRE_GUARD_BAND. Enforce this
139 * globally across all backends. */
140_Static_assert((ONEWIRE_ONE_PULSE + ONEWIRE_ZERO_PULSE + ONEWIRE_GUARD_BAND) < 256u,
141 "8-bit read capture (MSIZE=8) would truncate slot durations");

143/* --- OC3 output-compare preload (OC3PE) knob ---
144 * The preload bit in TIM_CCMR2 keeps the active CCR3 stable through a slot:
145 * with OC3PE the value written before CEN only enters the output at the
146 * terminal update event, which is the "line released to idle HIGH in
147 * hardware" guarantee (see README, "Bus Idle Behaviour"). The write/capture
148 * paths that rely on it (capture, read_pair, single-slot write) gate the bit
149 * behind this flag so a bench can sweep OC3PE on/off with
150 * -DOW_PORT_OC3PE=0/-DOW_PORT_OC3PE=1 without editing code. The DMA-fed paths
151 * (feed, write_then_read) leave it off unconditionally: their reload must act
152 * immediately, which preload would break.
153 *
154 * Default 1 (preload on) identical to the historical hard-coded behaviour.
155 *
156 * OW_PORT_OC3PE_ARGS (the CCMR2 bit value OR-ed into the mask at the affected
157 * call sites) is defined with the engine below, by the same #ifndef pattern
158 * as OW_PORT_IC4F_ARGS. */

174/* ============================================================
175 * The TIM1 + DMA core shared by all seven families
176 *
177 * The 1-Wire bus on all seven of these families is the same machine: TIM1 in
178 * one-pulse mode drives the slot pulse on CH3, CH4 captures the bus in
179 * indirect mode, and a plain CH2 compare at ONE+ZERO us is the end-of-slot
180 * marker that triggers the feed DMA into CCR3. Those 17 functions are one
181 * piece of code, not four that happen to look alike - they were four, because
182 * each backend was written against its own reference manual and then never
183 * merged.
184 *
185 * What actually differs between families is small enough to name: the clock
186 * gates in ow_port_init(), the bus-pin mux, and the DMA request routing.
187 * Clocks stay per-family (OW_PORT_ENABLE_BUS_CLOCKS), and so does the pin
188 * mux: OW_PORT_CONFIG_BUS_PIN() and OW_PORT_SET_PIN_MODE() are two short
189 * statement macros per backend (F1's legacy CRH field proves the shapes are
190 * family-owned, not shared). DMA request routing defaults to a no-op (fixed-map families) and G0/G4
191 * program their DMAMUX through the same two macros.
192 *
193 * The family facts above (ow_mcu.h: ow_pulse_t, OW_PORT_SYSCLK_MHZ /
194 * SystemCoreClock, OW_PORT_TIM1_UPD_IRQn, DMA spelling) must precede this
195 * core: the register vocabulary at the top of this file supplies the
196 * T1/PA/RC/D1x aliases used below (including the D13/D14 channel defaults),
197 * and the family section supplies ow_pulse_t and the init hooks.
198 * Concretely the core requires:
199 *
200 * - define OW_PORT_ENABLE_BUS_CLOCKS(), which gates this family's clocks;
201 * - define OW_PORT_CONFIG_BUS_PIN() and OW_PORT_SET_PIN_MODE(push_pull),
202 * which put the bus pin into alternate-function open-drain mode and
203 * toggle it to push-pull for the parasite strong pull-up;
204 * - optionally override OW_PORT_DMA_FEED / OW_PORT_DMA_CAPTURE (default
205 * D13/D14: TIM1_CC2 feed on channel 3, TIM1_CH4 capture on channel 4)
206 * and OW_PORT_ROUTE_CAPTURE() / OW_PORT_ROUTE_FEED() (default no-ops
207 * for fixed request maps; G0 programs its DMAMUX here). The channel
208 * assignment is the one thing here that is not interchangeable - read
209 * the family section's assignment against its reference manual.
210 *
211 * There is no run-time cost to the factoring: everything stays
212 * __STATIC_FORCEINLINE and every difference is resolved by the preprocessor,
213 * so a family's object code is what it was before the split. That is checked,
214 * not assumed - the .bin of every example for the families already sharing
215 * this core is byte-identical before and after the split that introduced it.
216 *
217 * F4 rides the same core through accessor macros with DMA1 defaults (see the
218 * OW_PORT_DMA_* block below) that the F4 section overrides — same bodies,
219 * family-spelled registers, with init clocks/pins/drive arriving through
220 * the OW_PORT_ENABLE_BUS_CLOCKS/OW_PORT_CONFIG_BUS_PIN hooks — and H5
221 * through GPDMA CTR1/CTR2/REQSEL vocabulary overrides the same way. Even the
222 * merged write+read pass is shared by all seven families with no
223 * per-family knobs left: two arm-order experiments on F446 (UG-vs-DMA order,
224 * early direction-pulse arm) both behave identically, so one order serves
225 * all - see the note above ow_port_write_then_read.
226 * ============================================================ */

272/* --- CH4 input-capture digital filter (IC4F): one selection for every clock,
273 * living here because the shared core is the one place all seven families
274 * are processed. The F4 section uses this same ladder through the core
275 * (it overrides only the DMA register spelling, not the timer setup), and
276 * so does H5 (GPDMA CTR1/CTR2/REQSEL overrides only).
277 *
278 * ICxF is a 4-bit ladder. The values this port actually programs, decoded
279 * from the RM0090 table (the same one ST's stm32f0xx_ll_tim.h spells out),
280 * are:
281 *
282 * ICxF_1 = 0b0001 = fCK_INT/16, N=1
283 * ICxF_0|ICxF_1 = 0b0011 = fCK_INT/4, N=1
284 * ICxF_0|ICxF_1|ICxF_2 = 0b0111 = fCK_INT/4, N=2
285 * ICxF_3 = 0b1000 = fCK_INT/32, N=3
286 * ICxF_0|ICxF_1|ICxF_3 = 0b1011 = fCK_INT/4, N=3
287 *
288 * So the >168MHz entry is fCK_INT/4 N=3, a *shorter* window than the
289 * 168MHz entry's fCK_INT/32 N=3. That is deliberate and was validated at
290 * 180MHz on the full 7-example matrix (see port/stm32f4/HARDWARE-NOTES.md).
291 * A "~500ns target" does not exist at these kernel clocks: the ladder's
292 * finest window is fCK_INT/16 (5.6ns at 180MHz), so reasoning should start
293 * from the decoded values.
294 *
295 * Backends feed the macro into TIM_CCMR2(...) unchanged. The default can
296 * be overridden from the build (-DOW_PORT_IC4F_ARGS=...) to sweep the
297 * filter on a bench.
298 * Test: tests/test/test_timing.c::test_ic4f_matches_the_documented_tier() */

300#ifndef OW_PORT_IC4F_ARGS

306#define OW_PORT_IC4F_ARGS IC4F_0, IC4F_1, IC4F_2 /* 0b0111 = fCK_INT/4, N=2 */
307#elif (OW_PORT_SYSCLK_MHZ) <= 168
308#define OW_PORT_IC4F_ARGS IC4F_3 /* 0b1000 = fCK_INT/32, N=3 */

335 * ow_mcu.h section: F0/F1/F3/G0/G4 leave the APB prescaler feeding TIM1 at
336 * /1, so TIM1 = PCLK = SYSCLK directly; F4 programs PPRE2=/2 (PPRE1=/4 or
337 * /2) and relies on the STM32 x2 doubling, so TIM1 = 2 x PCLK2 = SYSCLK
338 * either way. Which APB the timer sits on, and why that family's prescalers

347 * mode 2 on the bus pin, CC4 capturing the same pin through TI3, plus the
348 * IC4F tier above and the OC3PE knob. One macro so the three arms cannot
349 * drift; the DMA-fed merged arm drops the preload bit (see below). */

352#else /* !defined(OW_PORT_SYSCLK_MHZ): SystemCoreClock auto-path -------------
353 * For integrations that select a backend without passing a clock -D
354 * (PlatformIO/CubeMX family-macro path): the CMSIS SystemCoreClock variable
355 * supplies the frequency instead of a header fallback guess, so a CubeMX
356 * project at 64MHz can never silently inherit a 72MHz prescaler. The Make
357 * and CMake builds always pass -DOW_PORT_SYSCLK_MHZ (from chips/<part>.mk)

365extern uint32_t SystemCoreClock; /* CMSIS: maintained by SystemInit()/SystemCoreClockUpdate() */
366__STATIC_FORCEINLINE uint32_t ow_port_mhz_from_hz(uint32_t hz) {
367 /* Integer-MHz truncation of the system clock. Exact for every supported
368 * clock (8/16/48/64/72/84/168/170/180/250); shifts plus one 32x32
369 * multiply, so no __aeabi_uidiv on M0/M0+. The intermediate fits 32
370 * bits (524M at 250MHz). */
371 return (uint32_t)(((hz >> 8u) * 537u) >> 21u);
372}
373__STATIC_FORCEINLINE uint16_t ow_port_psc_from_hz(uint32_t hz) {
374 const uint32_t mhz = ow_port_mhz_from_hz(hz);
375 return (uint16_t)((mhz > 0u) ? (mhz - 1u) : 0u);
376}
377__STATIC_FORCEINLINE uint32_t ow_port_ic4f_mask_from_hz(uint32_t hz) {
378 /* Same tiers as the compile-time ladder above, as a numeric CCMR2 mask. */
379 const uint32_t mhz = ow_port_mhz_from_hz(hz);
380 if (mhz <= 8u) {
381 return TIM_CCMR2_IC4F_1;
382 }
383 if (mhz <= 16u) {
384 return (uint32_t)(TIM_CCMR2_IC4F_0 | TIM_CCMR2_IC4F_1);
385 }
386 if (mhz <= 72u) {
387 return (uint32_t)(TIM_CCMR2_IC4F_0 | TIM_CCMR2_IC4F_1 | TIM_CCMR2_IC4F_2);
388 }
389 if (mhz <= 168u) {
390 return TIM_CCMR2_IC4F_3;
391 }
392 return (uint32_t)(TIM_CCMR2_IC4F_0 | TIM_CCMR2_IC4F_1 | TIM_CCMR2_IC4F_3);
393}

425/* DMA disable + programming statement macros (port-unification vocabulary):
426 * the register-name level differs per DMA IP (DMA1 channels: CCR/CPAR/CMAR/
427 * CNDTR; F4 streams: CR/PAR/M0AR/NDTR + CHSEL inside the CR word above), so
428 * the bodies call these instead of spelling registers. Defaults are the
429 * DMA1 form; the F4 shim overrides DISABLE (stream CR + flag store) and the
430 * PROG pair (stream spelling). Pinned end to end by
431 * tests/test/test_dma_contract.c (per-operation CCR/CPAR/CMAR/CNDTR table). */
432#ifndef OW_PORT_DMA_EN_BIT
433#define OW_PORT_DMA_EN_BIT DMA_CCR_EN
434#endif
435#ifndef OW_PORT_DMA_DISABLE_CAPTURE
436#define OW_PORT_DMA_DISABLE_CAPTURE() ow_port_dma_disable(&OW_PORT_DMA_CAPTURE.CCR)
437#endif
438#ifndef OW_PORT_DMA_DISABLE_FEED
439#define OW_PORT_DMA_DISABLE_FEED() ow_port_dma_disable(&OW_PORT_DMA_FEED.CCR)
440#endif
441#ifndef OW_PORT_DMA_PROG_CAPTURE
442#define OW_PORT_DMA_PROG_CAPTURE(dst, count, cr) \
443 do { \
444 OW_PORT_DMA_CAPTURE.CPAR = (uint32_t)&T1.CCR4; \
445 OW_PORT_DMA_CAPTURE.CMAR = (uint32_t)(dst); \
446 OW_PORT_DMA_CAPTURE.CNDTR = (count); \
447 OW_PORT_DMA_CAPTURE.CCR = (cr); \
448 } while (0)
449#endif
450#ifndef OW_PORT_DMA_PROG_FEED
451#define OW_PORT_DMA_PROG_FEED(src, count, cr) \
452 do { \
453 OW_PORT_DMA_FEED.CPAR = (uint32_t)&T1.CCR3; \
454 OW_PORT_DMA_FEED.CMAR = (uint32_t)(src); \
455 OW_PORT_DMA_FEED.CNDTR = (count); \
456 OW_PORT_DMA_FEED.CCR = (cr); \
457 } while (0)

460/* Disable-path contract: the wait algorithm below is DMA-IP-agnostic (write
461 * 0, poll the EN bit to 0, bounded) and intentionally shared; each backend
462 * supplies only the operands (control register + EN bit) through its DISABLE
463 * macros and OW_PORT_DMA_EN_BIT. Duplicating the loop per backend would
464 * trade this single auditable copy for six drifting ones to hide register
465 * names that the PROG defaults already spell openly. */
488__STATIC_FORCEINLINE void ow_port_dma_disable(volatile uint32_t* ccr) {
489 *ccr = 0; /* request disable */
490 if (*ccr & OW_PORT_DMA_EN_BIT) {
491 uint32_t guard = 1000u;
492 while ((*ccr & OW_PORT_DMA_EN_BIT) != 0u && guard-- != 0u) {
493 }
494 }
495}

545__STATIC_FORCEINLINE uint8_t ow_port_bus_done(void) {
546 if (T1.SR & TIM_SR(UIF)) {
547 /* No software bus release needed: every operation returns the line to
548 * idle HIGH in hardware. DMA-fed writes (ow_port_feed,
549 * ow_port_write_then_read) append ONEWIRE_RELEASE_PULSE to the CCR3
550 * feed, and the direct-write/capture operations (reset, read, single
551 * slot) use an OC3PE preload of ONEWIRE_RELEASE_PULSE — both applied
552 * exactly when the one-pulse timer stops. */
553#if OW_PORT_LOW_POWER
554 /* The update event both interrupts the low-power WFE sleep and, via
555 * SEVONPEND, raises an NVIC pending bit. UIE also latches a pending
556 * bit at every ow_port_update_event() re-arm (EGR=UG). Clear the
557 * pending flag here so the next __WFE() truly sleeps; otherwise the
558 * pending bit would make __WFE() return immediately forever (silent
559 * degradation back to a busy-loop). Clearing UIF below also retires
560 * the long-stage condition read by ow_port_long_wait_pending(). */
561 NVIC_ClearPendingIRQ(OW_PORT_TIM1_UPD_IRQn);
562#endif
563 T1.SR = 0;
564 /* Retire the schedule explicitly: on hardware OPM already stopped the
565 * counter, but stating it here keeps the completion state honest
566 * without relying on that side effect (and keeps the host mock, which
567 * does not model OPM auto-stop, in agreement with hardware). */
568 T1.CR1 &= (uint32_t)~TIM_CR1(CEN);
569 return 1u;
570 }
571 return 0u;
572}

586__STATIC_FORCEINLINE uint8_t ow_port_long_wait_pending(void) {
587 if (!(T1.CR1 & TIM_CR1(CEN))) {
588 return 0u;
589 }
590 if (T1.SR & TIM_SR(UIF)) {
591 return 0u; /* armed but already complete: nothing left to sleep on */
592 }
593 return (uint8_t)(((uint32_t)T1.RCR + 1u) * (uint32_t)T1.ARR > OW_PORT_LONG_STAGE_US);
594}

602__STATIC_FORCEINLINE void ow_port_sleep_until_done(void) {
603 /* Prepare for sleep: clear any stale NVIC pending bit so a leftover
604 * event cannot wake the very first WFE (silent busy-loop degradation).
605 * A fresh pending bit will be latched by the timer's update when the
606 * long stage completes. */
607 NVIC_ClearPendingIRQ(OW_PORT_TIM1_UPD_IRQn);
608 /* Re-arm the event: SEV sets the event register, the first WFE returns
609 * immediately and clears it, so the second WFE in the loop truly sleeps
610 * until a new event arrives. */
611 __SEV();
612 __WFE();
613 while (!(T1.SR & TIM_SR(UIF))) {
614 __WFE(); /* sleeps; woken by the pending bit via SEVONPEND (no ISR) */
615 }
616 /* Consume the wake-up event so the next sleep starts from a clean state. */
617 NVIC_ClearPendingIRQ(OW_PORT_TIM1_UPD_IRQn);
618}

640__STATIC_FORCEINLINE void ow_port_capture(volatile void* dst, uint16_t count, uint16_t width) {
641#if OW_DRIVE_ACTIVE
642 ow_port_set_pin_mode(0); /* read/reset phases must be open-drain (slave can pull LOW) */
643#endif
644 T1.CCMR2 = OW_PORT_CCMR2_CAP;
645 T1.CCER = TIM_CCER(CC3E, CC4E);
646#if OW_PORT_LOW_POWER
647 T1.DIER = TIM_DIER(CC4DE, UIE);
648 /* No duration decision here: ow_port_long_wait_pending() derives it from
649 * the caller's ARR/RCR, so every capture path (reset, read, long
650 * scratchpad read) shares the one OW_PORT_LONG_STAGE_US rule. */
651#else
652 T1.DIER = TIM_DIER(CC4DE);
653#endif
654 ow_port_update_event();
655 T1.CCR3 = ONEWIRE_RELEASE_PULSE;
656 OW_PORT_DMA_DISABLE_CAPTURE();
657 OW_PORT_ROUTE_CAPTURE();
658 OW_PORT_DMA_PROG_CAPTURE(dst, count, ((width == 16) ? OW_PORT_DMA_CR_RX16 : OW_PORT_DMA_CR_RX8));
659 T1.CR1 = TIM_CR1(OPM, CEN);
660}

676__STATIC_FORCEINLINE uint8_t ow_port_feed(const ow_pulse_t* cmd, uint16_t slots) {
677 if (slots == 0u || slots > ONEWIRE_MAX_SLOTS) {
678 return 0u;
679 }
680 T1.RCR = slots - 1;
681 T1.ARR = ONEWIRE_ONE_PULSE + ONEWIRE_ZERO_PULSE + ONEWIRE_GUARD_BAND;
682 T1.CCR3 = cmd[0];
683 T1.CCR2 = ONEWIRE_ONE_PULSE + ONEWIRE_ZERO_PULSE;
684 T1.CCMR2 = TIM_CCMR2(OC3M_0, OC3M_1, OC3M_2);
685 T1.CCER = TIM_CCER(CC3E);
686#if OW_PORT_LOW_POWER
687 T1.DIER = TIM_DIER(CC2DE, UIE);
688#else
689 T1.DIER = TIM_DIER(CC2DE);
690#endif
691 ow_port_update_event();
692 OW_PORT_DMA_DISABLE_FEED();
693 OW_PORT_ROUTE_FEED();
694 /* Feed slots 2..N, then the ONEWIRE_RELEASE_PULSE (bus release) */
695 OW_PORT_DMA_PROG_FEED(&cmd[1], slots, OW_PORT_DMA_CR_TX);
696 T1.CR1 = TIM_CR1(OPM, CEN);
697 return 1u;
698}

705__STATIC_FORCEINLINE void ow_port_start_timer(uint16_t arr, uint8_t rcr) {
706 T1.ARR = arr;
707 T1.RCR = rcr;
708#if OW_PORT_LOW_POWER
709 if ((uint32_t)(rcr + 1u) * (uint32_t)arr > OW_PORT_LONG_STAGE_US) {
710 /* long stage: conversion / EEPROM hold-off. Enable the update
711 * interrupt so the pending bit wakes __WFE() via SEVONPEND.
712 * ow_port_capture() already sets UIE for the capture stages, but
713 * start_timer() must too, else WFE sleeps forever. */
714 T1.DIER |= TIM_DIER(UIE);
715 }
716#endif
717 ow_port_update_event();
718 T1.CR1 = TIM_CR1(OPM, CEN);
719}

726__STATIC_FORCEINLINE void ow_port_reset(volatile uint16_t* reset_pulses) {
727 T1.RCR = 0;
728 T1.ARR = OW_PORT_RESET_TIMEOUT;
729 T1.CCR3 = OW_PORT_RESET_PULSE_DURATION;
730 /* Clear the capture buffer: only reset_pulses[0] (master release) is always
731 * written by the DMA, so a no-presence reset would otherwise leave a stale
732 * reset_pulses[1] from a previous presence reset and onewire_present() would
733 * report a false device. Zeroing makes a single-capture reset report "no
734 * device". */
735 reset_pulses[0] = 0;
736 reset_pulses[1] = 0;
737 ow_port_capture(reset_pulses, OW_PORT_CAPTURE_BUF_SIZE, 16);
738}

750__STATIC_FORCEINLINE uint8_t ow_port_write_slots(const ow_pulse_t* pulses, uint16_t slots) {
751 if (slots == 0u || slots > ONEWIRE_MAX_SLOTS) {
752 return 0u;
753 }
754#if OW_DRIVE_ACTIVE
755 ow_port_set_pin_mode(1); /* active-drive write: master drives both levels */
756#endif
757 if (slots == 1) {
758 /* Single slot: no DMA needed, avoids a zero-length DMA transaction */
759 T1.RCR = 0; /* Single slot, no repetition */
760 T1.ARR = ONEWIRE_ONE_PULSE + ONEWIRE_ZERO_PULSE + ONEWIRE_GUARD_BAND; /* Total bit slot time */
761 T1.CCR3 = pulses[0]; /* Pulse duration encodes the bit */
762 /* OC3PE plus a ONEWIRE_RELEASE_PULSE preload release the bus at the
763 * terminal update event, exactly when the one-pulse timer stops
764 * (hardware bus release). */
765 T1.CCMR2 = TIM_CCMR2(OC3M_0, OC3M_1, OC3M_2) | OW_PORT_OC3PE_ARGS;
766 T1.CCER = TIM_CCER(CC3E);
767#if OW_PORT_LOW_POWER
768 T1.DIER = TIM_DIER(UIE); /* no DMA for a single bit slot; keep UIE for WFE */
769#else
770 T1.DIER = 0; /* No DMA for a single bit slot */
771#endif
772 ow_port_update_event();
773 T1.CCR3 = ONEWIRE_RELEASE_PULSE; /* Preload release pulse -> line idles HIGH when the timer stops */
774 T1.CR1 = TIM_CR1(OPM, CEN);
775 return 1u;
776 }
777 return ow_port_feed(pulses, slots);
778}

803/* Merged-pass arm order: one order serves all families (direction pulse
804 * armed first, DMA programmed after the update event). Two per-family knobs
805 * lived here briefly and were both removed after full F446 hardware matrices
806 * proved them don't-care: DMA-after-UG behaves like DMA-before-UG, and the
807 * early direction-pulse arm behaves like arming once before CEN (14 cells
808 * each, 2026-10-06). */
816__STATIC_FORCEINLINE void ow_port_write_then_read(uint8_t bit, volatile uint16_t* pulse3,
817 const ow_pulse_t* read_pulse) {
818#if OW_DRIVE_ACTIVE
819 ow_port_set_pin_mode(0); /* merged write+read stays open-drain so the read half is safe */
820#endif
821 const ow_pulse_t write_pulse = bit ? ONEWIRE_ONE_PULSE : ONEWIRE_ZERO_PULSE;
822 OW_PORT_MARKER_TOGGLE(); /* opt-in LA hook (no-op by default): merged pass starts here */
823 T1.RCR = 2; /* Three slots, then a single update event */
824 T1.ARR = ONEWIRE_ONE_PULSE + ONEWIRE_ZERO_PULSE + ONEWIRE_GUARD_BAND; /* Total bit slot time */
825 /* Arm the direction pulse first. The bus was released idle-high by
826 * ow_port_bus_done(), so this write produces the single clean falling edge
827 * the devices re-sync their slot timer to. Holding it from the top instead
828 * of arming it right before CEN means the CC4 capture is armed while the
829 * bus is low, so the open-drain RC rise can never be mistaken for a slot
830 * edge. */
831 T1.CCR3 = write_pulse; /* Slot 1 write pulse encodes the direction bit */
832 T1.CCR2 = ONEWIRE_ONE_PULSE + ONEWIRE_ZERO_PULSE; /* End-of-slot reload trigger */
833 /* OC3 in PWM mode (no preload so the reload is immediate), CC4 capture armed */
834 T1.CCMR2 = OW_PORT_CCMR2_MRG;
835 T1.CCER = TIM_CCER(CC3E, CC4E); /* Enable both channels */
836 /* Disconnect DMA requests while re-arming the channels, then re-connect
837 * them only after the timer flags are clean and just before starting.
838 * (The end-of-slot CC2 compare event of the previous merged operation can
839 * leave a pending request that fires the reload DMA immediately on re-arm,
840 * overwriting the freshly written direction pulse in CCR3.) */
841#if OW_PORT_LOW_POWER
842 T1.DIER = TIM_DIER(UIE); /* keep UIE for WFE while the DMA requests are apart */
843#else
844 T1.DIER = 0;
845#endif
846 ow_port_update_event();
847 /* Capture DMA: write-slot capture plus the id/cmp pulse pair into the buffer.
848 * Spelled through the PROG vocabulary (DMA1 defaults above write the same
849 * CPAR/CMAR/CNDTR/CCR sequence), so a backend on different DMA IP only
850 * overrides the macros, not this body. */
851 OW_PORT_DMA_DISABLE_CAPTURE();
852 OW_PORT_ROUTE_CAPTURE();
853 OW_PORT_DMA_PROG_CAPTURE(pulse3, 3, OW_PORT_DMA_CR_RX16);
854 /* Feed DMA: reload CCR3 with the read pulse for slots 2-3, then write
855 * ONEWIRE_RELEASE_PULSE during slot 3 so the one-pulse timer stops with
856 * the line released to idle HIGH (hardware bus release). */
857 OW_PORT_DMA_DISABLE_FEED();
858 OW_PORT_ROUTE_FEED();
859 OW_PORT_DMA_PROG_FEED(read_pulse, 3, OW_PORT_DMA_CR_TX);
860#if OW_PORT_LOW_POWER
861 T1.DIER = TIM_DIER(CC4DE, CC2DE, UIE); /* Capture + CCR3 reload via DMA (UIE for WFE) */
862#else
863 T1.DIER = TIM_DIER(CC4DE, CC2DE); /* Capture + CCR3 reload via DMA */
864#endif
865 T1.CCR3 = write_pulse; /* Re-arm the direction pulse (safe against a stale CC2 DMA reload) */
866 T1.CR1 = TIM_CR1(OPM, CEN);
867}
