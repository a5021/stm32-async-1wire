97/* @brief PA10: alternate function, open-drain, AF2 (TIM1_CH3).
98 *
99 * F0 has the modern GPIO register model, as G0 does; F1 configures the same
100 * pin through the legacy CRH field instead, which is why this is a macro
101 * rather than shared code. The whole MODE field is cleared first so the
102 * pin lands in the right mode even if something set it before us.
103 */
104#define OW_PORT_CONFIG_BUS_PIN() \
105 do { \
106 PA.MODER = (PA.MODER & ~GPIO_MODER_MODER10) | GPIO_MODER_MODER10_1; \
107 PA.OTYPER |= GPIO_OTYPER_OT_10; \
108 PA.AFR[1] = (PA.AFR[1] & ~GPIO_AFRH_AFSEL10) | (2u << GPIO_AFRH_AFSEL10_Pos); \
109 /* Drive strength is configurable via OW_BUS_DRIVE, default MAX. */ \
110 PA.OSPEEDR = (PA.OSPEEDR & ~GPIO_OSPEEDR_OSPEEDR10) | \
111 ((OW_BUS_DRIVE & 0x3u) << GPIO_OSPEEDR_OSPEEDR10_Pos); \
112 } while (0)
